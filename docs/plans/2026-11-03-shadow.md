# shadow: Implementation Plan

**Spec:** docs/specs/2026-11-03-shadow-design.md
**Goal:** Elements get a soft drop shadow through `.custom = Shadow(...)`, drawn by the raylib renderer with stacked rounded rectangles.
**Architecture:** `CC_Shadow` fills a slot in a per-frame pool and returns it as the element's `.custom.customData`, tagged `CC_SHADOW_TAG (-2)`. Clay emits that CUSTOM command before the element's background RECTANGLE. The raylib renderer recognises the tag and draws 8 fading rounded rectangles (or one when blur is 0) without a scissor, then lets the background draw as usual.

## Global constraints
- Must work on Windows, Linux, macOS and the Emscripten web build without custom GLSL.
- `external/clay` and `include/clay.h` are not edited.
- C11, 4 space indent, `CC_*` public names, `cc__*` file local names.
- Files CI format-checks (`src/`, `include/ccompose.h`, `examples/`, `tests/`) must pass the repository `.clang-format`. CI runs clang-format 22.

/* Shadow — a drop shadow behind an element, set through its .custom
 * field. The shadow follows the element's own .cornerRadius.
 *
 *     Column("card", .backgroundColor = ColorHex(0x2A2D34),
 *            .cornerRadius = RadiusAll(8),
 *            .custom = Shadow(.offset = {0, 4}, .blur = 12)) { ... }
 *
 * The raylib renderer fakes the blur with 8 rounded rectangles that grow
 * and fade, so it needs no shader and works on desktop and web. Other
 * renderers see a CUSTOM command tagged CC_SHADOW_TAG and may skip it.
 *
 * One element holds one .custom value, so an element can't have both a
 * shadow and a Draw callback. Wrap the drawn element in a parent that
 * carries the shadow instead.
 *
 * Shadow data lives in a per-frame pool of CC_SHADOW_POOL_SIZE entries,
 * reset by CC_Begin(). When the pool is full Shadow() returns
 * {.customData = NULL} and the element has no shadow. */
typedef struct {
  CC_Vector2 offset; /* shift in px, {0, 4} drops the shadow down. */
  float blur;        /* soft edge width in px. 0 = hard edge. Negative = 0. */
  float spread;      /* grows (+) or shrinks (-) the shadow before blur. */
  CC_Color color;    /* alpha 0 means the default Color(0, 0, 0, 64). */
} CC_ShadowOpts;

/* What a Shadow() customData points to. The renderer checks _tag. */
#define CC_SHADOW_TAG (-2)
#define CC_SHADOW_POOL_SIZE 128
typedef struct {
  int _tag;
  CC_ShadowOpts opts;
} CC_ShadowSlot;

CC_CustomElementConfig CC_Shadow(CC_ShadowOpts opts);

#define Shadow(...) CC_Shadow((CC_ShadowOpts){__VA_ARGS__})
```

/* Per-frame pool behind Shadow(). Lives outside the backend #ifdef so
 * headless tests can inspect the emitted CUSTOM commands. */
static CC_ShadowSlot cc__shadow_pool[CC_SHADOW_POOL_SIZE];
static int cc__shadow_pool_used = 0;

CC_CustomElementConfig CC_Shadow(CC_ShadowOpts opts) {
  if (cc__shadow_pool_used >= CC_SHADOW_POOL_SIZE)
    return (CC_CustomElementConfig){0};
  if (opts.blur < 0.0f)
    opts.blur = 0.0f;
  if (opts.color.a == 0)
    opts.color = (CC_Color){0, 0, 0, 64};
  CC_ShadowSlot *s = &cc__shadow_pool[cc__shadow_pool_used++];
  s->_tag = CC_SHADOW_TAG;
  s->opts = opts;
  return (CC_CustomElementConfig){.customData = s};
}
```

/* Headless test - Shadow.
 *
 * Invariants:
 *   1. An element with .custom = Shadow(...) emits one CUSTOM command
 *      whose data carries CC_SHADOW_TAG and the passed values, before
 *      the element's background RECTANGLE.
 *   2. Alpha 0 falls back to {0, 0, 0, 64}, negative blur becomes 0.
 *   3. The pool holds CC_SHADOW_POOL_SIZE shadows per frame, the next
 *      call returns NULL, and CC_Begin() empties the pool.
 * */

#include <assert.h>
#include <stdbool.h>
#include <stdio.h>

#include "ccompose.h"

static bool saw_error = false;

static void err_handler(CC_ErrorData err) {
  saw_error = true;
  fprintf(stderr, "clay error: %.*s\n", (int)err.errorText.length,
          err.errorText.chars);
}

static bool same_color(CC_Color a, CC_Color b) {
  return a.r == b.r && a.g == b.g && a.b == b.b && a.a == b.a;
}

int main(void) {
  CC_SetViewport(400.0f, 400.0f);
  CC_SetErrorHandler(err_handler);
  CC_Init();

  /* 1 and 2 */
  CC_Begin();
  CC_CustomElementConfig full =
      Shadow(.offset = {3, 4}, .blur = 12, .spread = 2,
             .color = Color(1, 2, 3, 90));
  CC_CustomElementConfig dflt = Shadow(.blur = -5);
  Column("Root", .layout = {.sizing = {Grow(), Grow()}}) {
    Column("Card", .layout = {.sizing = {Fixed(100), Fixed(50)}},
           .backgroundColor = Color(200, 200, 200, 255),
           .cornerRadius = RadiusAll(8), .custom = full) {}
  }
  CC_RenderCommandArray cmds = CC_End();
  assert(!saw_error);

  CC_ShadowSlot *s = (CC_ShadowSlot *)full.customData;
  assert(s && s->_tag == CC_SHADOW_TAG);
  assert(s->opts.offset.x == 3 && s->opts.offset.y == 4);
  assert(s->opts.blur == 12 && s->opts.spread == 2);
  assert(same_color(s->opts.color, Color(1, 2, 3, 90)));

  CC_ShadowSlot *d = (CC_ShadowSlot *)dflt.customData;
  assert(d && d->_tag == CC_SHADOW_TAG);
  assert(d->opts.blur == 0);
  assert(same_color(d->opts.color, Color(0, 0, 0, 64)));

  uint32_t card = CC_SID(CC__Str("Card")).id;
  int custom_at = -1, rect_at = -1, customs = 0;
  for (int32_t i = 0; i < cmds.length; ++i) {
    CC_RenderCommand *c = CC_RenderCommandArray_Get(&cmds, i);
    if (c->id != card)
      continue;
    if (c->commandType == CLAY_RENDER_COMMAND_TYPE_CUSTOM) {
      customs++;
      custom_at = i;
      assert(c->renderData.custom.customData == s);
      assert(c->renderData.custom.cornerRadius.topLeft == 8);
    }
    if (c->commandType == CLAY_RENDER_COMMAND_TYPE_RECTANGLE)
      rect_at = i;
  }
  assert(customs == 1);
  assert(rect_at >= 0 && custom_at < rect_at);

  /* 3 */
  CC_Begin();
  for (int i = 0; i < CC_SHADOW_POOL_SIZE; ++i)
    assert(Shadow(.blur = 1).customData != NULL);
  assert(Shadow(.blur = 1).customData == NULL);
  CC_End();
  CC_Begin();
  assert(Shadow(.blur = 1).customData != NULL);
  CC_End();

  assert(!saw_error);
  CC_Shutdown();
  printf("shadow_test: ok\n");
  return 0;
}
```

If `CC_SID` or `CC__Str` usage does not compile, copy the id lookup form from `tests/progress_test.c` (`box_of`) or `tests/box_test.c`. The asserts stay as written.

/* Paints a Shadow() slot behind its element. The blur is faked with
 * stacked rounded rectangles so it needs no shader. Each layer's alpha
 * is chosen so the fully overlapped middle composites to the requested
 * alpha. No scissor: the shadow is meant to reach outside the box.
 * vibekit: Clay culls fully off-screen elements, so their shadow is not
 * drawn even when it would reach on screen. Emit the shadow as its own
 * floating root if that ever shows. */
#define CC_SHADOW_LAYERS 8
static void cc__paint_shadow(Clay_BoundingBox bb, float radius,
                             const CC_ShadowSlot *slot) {
  CC_ShadowOpts s = slot->opts;
  Rectangle base = {bb.x + s.offset.x - s.spread, bb.y + s.offset.y - s.spread,
                    bb.width + 2 * s.spread, bb.height + 2 * s.spread};
  float base_radius = radius + s.spread > 0 ? radius + s.spread : 0;
  int layers = s.blur > 0 ? CC_SHADOW_LAYERS : 1;
  float a = s.color.a / 255.0f;
  float layer_a = layers == 1 ? a : 1.0f - powf(1.0f - a, 1.0f / layers);
  Color c = {(unsigned char)s.color.r, (unsigned char)s.color.g,
             (unsigned char)s.color.b, (unsigned char)(layer_a * 255.0f + 0.5f)};
  for (int i = layers; i >= 1; --i) {
    float grow = layers == 1 ? 0 : s.blur * i / layers;
    Rectangle r = {base.x - grow, base.y - grow, base.width + 2 * grow,
                   base.height + 2 * grow};
    if (r.width <= 0 || r.height <= 0)
      continue;
    float shorter = r.width < r.height ? r.width : r.height;
    float rad = base_radius > 0 ? base_radius + grow : 0;
    if (rad > 0)
      DrawRectangleRounded(r, rad * 2 / shorter, 8, c);
    else
      DrawRectangleRec(r, c);
  }
}
```
